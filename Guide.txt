تصمیم بسیار هوشمندانه‌ای گرفتید! اجرای هر سناریو روی یک ماشین مجازی (VM) مجزا و ایزوله با مشخصات سخت‌افزاری کاملاً یکسان، بهترین رویکرد علمی برای حذف اثرات جانبی (Side Effects) و تداخل کش حافظه و پروسه‌هاست.

در ادامه، دستورالعمل جامع، بدون نقص و خط‌به‌خط (End-to-End) از کلون کردن سورس OVS از گیت‌هاب تا جایگذاری دقیق فایل‌ها، کامپایل، راه‌اندازی شبکه مجازی، تولید بار و خروجی CSV برای هر ۴ ماشین مجازی آورده شده است.

ماتریس ماشین‌ها و سناریوها
VM-1: سناریوی BASE (نسخه رسمی OVS بدون تغییر)
VM-2: سناریوی LOCAL (فیلتر کوکو محلی روی هر ساب‌تیبل)
VM-3: سناریوی GLOBAL_AI (فیلتر گلوبال + هوش مصنوعی)
VM-4: سناریوی FULL_TIER (معماری کامل سه‌لایه‌ای: AI + Global + Local)
گام مشترک در هر ۴ ماشین مجازی (Setup & Prerequisites)
در ترمینال هر ۴ ماشین مجازی دستورات زیر را کپی و اجرا کنید:

# ۴. ساخت فایل اسکریپت تولید ترافیک قطعی (traffic_gen.py)
cat << 'EOF' > traffic_gen.py
import struct
import socket

PCAP_FILE = "benchmark_traffic.pcap"
NUM_PACKETS = 100000

def write_pcap_header(f):
    header = struct.pack("!IHHiIII", 0xa1b2c3d4, 2, 4, 0, 0, 65535, 1)
    f.write(header)

def make_eth_ip_udp(src_ip, dst_ip, src_port, dst_port, payload=b"BENCHMARK_PKT"):
    eth = b"\x00\x00\x00\x00\x00\x02\x00\x00\x00\x00\x00\x01\x08\x00"
    ip_len = 20 + 8 + len(payload)
    ip_hdr = struct.pack(
        "!BBHHHBBH4s4s",
        0x45, 0, ip_len, 54321, 0, 64, 17, 0,
        socket.inet_aton(src_ip), socket.inet_aton(dst_ip)
    )
    udp_hdr = struct.pack("!HHHH", src_port, dst_port, 8 + len(payload), 0)
    return eth + ip_hdr + udp_hdr + payload

def main():
    print(f"[*] Generating {NUM_PACKETS} deterministic packets into {PCAP_FILE}...")
    with open(PCAP_FILE, "wb") as f:
        write_pcap_header(f)
        for i in range(NUM_PACKETS):
            if i % 10 < 8:
                cat = i % 5
                if cat == 0:
                    pkt = make_eth_ip_udp("10.0.0.1", "10.0.0.2", 5001, 5001)
                elif cat == 1:
                    pkt = make_eth_ip_udp("10.0.0.1", "10.0.0.2", 1000 + (i % 20), 5002)
                elif cat == 2:
                    pkt = make_eth_ip_udp(f"10.0.1.{(i % 250) + 1}", "10.0.0.2", 2000, 6000)
                elif cat == 3:
                    pkt = make_eth_ip_udp("10.0.0.1", "10.0.0.2", 3000, 7000 + (i % 10))
                else:
                    pkt = make_eth_ip_udp("10.0.0.1", "10.0.0.2", 4000, 8000)
            else:
                pkt = make_eth_ip_udp(f"192.168.{(i % 250) + 1}.1", "172.16.0.2", 9000, 9999)

            ts_sec = i // 10000
            ts_usec = (i % 10000) * 100
            pkt_len = len(pkt)
            pkt_hdr = struct.pack("!IIII", ts_sec, ts_usec, pkt_len, pkt_len)
            f.write(pkt_hdr + pkt)
    print(f"[+] PCAP ready: {PCAP_FILE}")

if __name__ == "__main__":
    main()
EOF

# ۵. تولید فایل ترافیک (۱۰۰ هزار پکت یکسان در هر ۴ سرور)
python3 traffic_gen.py

# ۶. ساخت فایل بنچ‌مارک و خروجی CSV (run_benchmark.py)
cat << 'EOF' > run_benchmark.py
import sys
import time
import subprocess
import argparse
import csv
import re
import psutil

def get_ovs_pid():
    for p in psutil.process_iter(['name', 'cmdline']):
        try:
            if 'ovs-vswitchd' in p.info['name'] or (p.info['cmdline'] and any('ovs-vswitchd' in c for c in p.info['cmdline'])):
                return p.pid
        except (psutil.NoSuchProcess, psutil.AccessDenied):
            continue
    return None

def clear_ovs_stats():
    subprocess.run(["ovs-appctl", "dpif-netdev/pmd-stats-clear"], stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)

def get_pmd_stats():
    res = subprocess.run(["ovs-appctl", "dpif-netdev/pmd-stats-show"], capture_output=True, text=True)
    out = res.stdout
    rx = tx = megaflow_hits = lookups_per_hit = 0
    
    rx_m = re.search(r"Rx packets:\s+(\d+)", out)
    tx_m = re.search(r"Tx packets:\s+(\d+)", out)
    mf_m = re.search(r"Megaflow hits:\s+(\d+)\s+\([^\)]+,\s*([\d\.]+)\s+subtbl lookups/hit\)", out)
    
    if rx_m: rx = int(rx_m.group(1))
    if tx_m: tx = int(tx_m.group(1))
    if mf_m:
        megaflow_hits = int(mf_m.group(1))
        lookups_per_hit = float(mf_m.group(2))
    return rx, tx, megaflow_hits, lookups_per_hit

def run_test(scenario, duration_sec=15):
    print(f"=== Starting Benchmark for Scenario: {scenario} ===")
    ovs_pid = get_ovs_pid()
    if not ovs_pid:
        print("[!] Error: ovs-vswitchd is not running.")
        sys.exit(1)
    
    proc = psutil.Process(ovs_pid)
    clear_ovs_stats()
    
    print("[*] Replaying PCAP traffic in ns1...")
    pcap_cmd = "tcpreplay --topspeed --loop=3 -i veth1 benchmark_traffic.pcap"
    replay_proc = subprocess.Popen(
        ["ip", "netns", "exec", "ns1", "bash", "-c", pcap_cmd],
        stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL
    )
    
    cpu_records = []
    mem_records = []
    start_time = time.time()
    
    while time.time() - start_time < duration_sec:
        cpu = proc.cpu_percent(interval=0.2)
        mem = proc.memory_info().rss / (1024 * 1024)
        cpu_records.append(cpu)
        mem_records.append(mem)
        if replay_proc.poll() is not None:
            break
            
    try:
        replay_proc.kill()
    except:
        pass
        
    rx, tx, mf_hits, lookups_per_hit = get_pmd_stats()
    avg_cpu = sum(cpu_records) / len(cpu_records) if cpu_records else 0
    max_cpu = max(cpu_records) if cpu_records else 0
    avg_mem = sum(mem_records) / len(mem_records) if mem_records else 0
    elapsed = max(time.time() - start_time, 0.001)
    mpps = (rx / elapsed) / 1e6
    
    ping_res = subprocess.run(
        ["ip", "netns", "exec", "ns1", "ping", "-c", "20", "-i", "0.2", "10.0.0.2"],
        capture_output=True, text=True
    )
    rtt_match = re.search(r"rtt min/avg/max/mdev = ([\d\.]+)/([\d\.]+)/([\d\.]+)/([\d\.]+)", ping_res.stdout)
    rtt_avg = float(rtt_match.group(2)) if rtt_match else 0.0

    result = {
        "Scenario": scenario,
        "Total_Rx_Packets": rx,
        "Total_Tx_Packets": tx,
        "Throughput_Mpps": round(mpps, 4),
        "Subtbl_Lookups_per_Hit": lookups_per_hit,
        "Megaflow_Hits": mf_hits,
        "Avg_CPU_Percent": round(avg_cpu, 2),
        "Max_CPU_Percent": round(max_cpu, 2),
        "Avg_Memory_MB": round(avg_mem, 2),
        "Avg_RTT_Latency_ms": round(rtt_avg, 3)
    }
    
    filename = f"results_{scenario.lower()}.csv"
    with open(filename, "w", newline="") as f:
        writer = csv.DictWriter(f, fieldnames=result.keys())
        writer.writeheader()
        writer.writerow(result)
        
    print(f"[+] Benchmark completed! Results saved to: {filename}")
    print(result)

if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    parser.add_argument("--scenario", required=True, choices=["BASE", "LOCAL", "GLOBAL_AI", "FULL_TIER"])
    args = parser.parse_args()
    run_test(args.scenario)
EOF
دستورالعمل اختصاصی ماشین ۱ (VM-1: Baseline)
در این ماشین، نسخه رسمی و بدون تغییر OVS را از گیت‌هاب کلون و تست می‌کنیم:

bash
cd ~
# کلون آخرین تگ پایدار یا برنچ رسمی
git clone https://github.com/openvswitch/ovs.git ovs-base
cd ovs-base
git checkout v3.2.0 || git checkout master

# ۱. بیلد و کامپایل OVS خام
./boot.sh
./configure --prefix=/usr --localstatedir=/var --sysconfdir=/etc
make -j$(nproc)
sudo make install

# ۲. ایجاد دیتابیس تمیز و اجرای سرویس OVS
sudo rm -rf /etc/openvswitch/conf.db
sudo ovsdb-tool create /etc/openvswitch/conf.db /usr/share/openvswitch/vswitch.ovsschema

sudo ovsdb-server --remote=punix:/var/run/openvswitch/db.sock \
    --remote=db:Open_vSwitch,Open_vSwitch,manager_options \
    --pidfile=/var/run/openvswitch/ovsdb-server.pid --detach

sudo ovs-vsctl --no-wait init

sudo ovs-vswitchd --pidfile=/var/run/openvswitch/ovs-vswitchd.pid --detach \
    --log-file=/var/log/openvswitch/ovs-vswitchd.log

# ۳. اتصال پورت‌ها به بریج netdev
sudo ovs-vsctl add-br br-test -- set bridge br-test datapath_type=netdev
sudo ovs-vsctl add-port br-test ovs-veth1
sudo ovs-vsctl add-port br-test ovs-veth2

# ۴. اعمال فلوهای استاندارد آزمون
sudo ovs-ofctl del-flows br-test
sudo ovs-ofctl add-flow br-test "priority=100,ip,nw_src=10.0.0.1,nw_dst=10.0.0.2,tp_dst=5001 actions=output:ovs-veth2"
sudo ovs-ofctl add-flow br-test "priority=95,ip,nw_src=10.0.0.1,nw_dst=10.0.0.2,udp,tp_dst=5002 actions=output:ovs-veth2"
sudo ovs-ofctl add-flow br-test "priority=90,ip,nw_src=10.0.0.0/24,udp actions=output:ovs-veth2"
sudo ovs-ofctl add-flow br-test "priority=85,ip,nw_dst=10.0.0.2,udp,tp_dst=6000 actions=output:ovs-veth2"
sudo ovs-ofctl add-flow br-test "priority=80,ip,nw_src=10.0.0.1,tp_dst=7000 actions=output:ovs-veth2"
sudo ovs-ofctl add-flow br-test "priority=100,in_port=ovs-veth2 actions=output:ovs-veth1"
sudo ovs-ofctl add-flow br-test "priority=10 actions=NORMAL"

# ۵. اجرای بنچ‌مارک سناریو ۱
cd ~
sudo python3 run_benchmark.py --scenario BASE
